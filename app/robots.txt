import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: [
        '/dashboard/', 
        '/api/', 
        '/prihlaseni', 
        '/registrace',
        '/rezervace'
      ],
    },
    sitemap: 'https://emsznojmo.cz/sitemap.xml',
  }
}
